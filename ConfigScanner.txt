private static X509Certificate2 GetCertificate(string thumbprint)
{
    using (var store = new X509Store(StoreName.My, StoreLocation.LocalMachine))
    {
        store.Open(OpenFlags.ReadOnly);

        var certs = store.Certificates.Find(
            X509FindType.FindByThumbprint,
            thumbprint,
            false
        );

        if (certs.Count == 0)
            throw new Exception("Certificate not found in LocalMachine\\My");

        return certs[0];
    }
}

using Microsoft.Identity.Client;
using System;
using System.Security.Cryptography.X509Certificates;

public static string TestToken(string clientId, string tenantId, X509Certificate2 certificate, string webUrl)
{
    var uri = new Uri(webUrl);
    var scopes = new[] { $"{uri.Scheme}://{uri.Authority}/.default" };

    var app = ConfidentialClientApplicationBuilder
        .Create(clientId)
        .WithTenantId(tenantId)
        .WithCertificate(certificate)
        .WithLogging(
            (level, message, containsPii) =>
            {
                Utils.WritetoEventViewer($"MSAL {level}: {message}");
            },
            LogLevel.Verbose,
            enablePiiLogging: true,
            enableDefaultPlatformLogging: false)
        .Build();

    Utils.WritetoEventViewer("Before AcquireTokenForClient");

    var result = app
        .AcquireTokenForClient(scopes)
        .ExecuteAsync()
        .GetAwaiter()
        .GetResult();

    Utils.WritetoEventViewer("After AcquireTokenForClient");
    Utils.WritetoEventViewer("Token acquired");

    return result.AccessToken;
}

namespace ConfigScanner
{
    internal class Program
    {
        static void Main(string[] args)
        {
            string[] rootPaths =
            {
                @"E:\TAG\IN",
                @"E:\TAG\OUT"
            };

            string outputCsvPath = Path.Combine(
                AppDomain.CurrentDomain.BaseDirectory,
                $"ConfigScanResults_{DateTime.Now:yyyyMMdd_HHmmss}.csv");

            List<ConfigResult> results = new List<ConfigResult>();

            foreach (string rootPath in rootPaths)
            {
                if (!Directory.Exists(rootPath))
                {
                    results.Add(new ConfigResult
                    {
                        RootPath = rootPath,
                        Mnemonic = "",
                        ConfigPath = "",
                        IsSPOOnline = "",
                        AppID = "",
                        Status = "Root path not found",
                        ErrorMessage = ""
                    });
                    continue;
                }

                string[] subFolders;
                try
                {
                    subFolders = Directory.GetDirectories(rootPath);
                }
                catch (Exception ex)
                {
                    results.Add(new ConfigResult
                    {
                        RootPath = rootPath,
                        Mnemonic = "",
                        ConfigPath = "",
                        IsSPOOnline = "",
                        AppID = "",
                        Status = "Failed to read subfolders",
                        ErrorMessage = ex.Message
                    });
                    continue;
                }

                foreach (string subFolder in subFolders)
                {
                    string mnemonic = Path.GetFileName(subFolder);
                    string pgmPath = Path.Combine(subFolder, "PGM");
                    string configPath = Path.Combine(pgmPath, "app.config");

                    ConfigResult result = new ConfigResult
                    {
                        RootPath = rootPath,
                        Mnemonic = mnemonic,
                        ConfigPath = configPath,
                        IsSPOOnline = "",
                        AppID = "",
                        Status = "",
                        ErrorMessage = ""
                    };

                    try
                    {
                        if (!Directory.Exists(pgmPath))
                        {
                            result.Status = "PGM folder not found";
                            results.Add(result);
                            continue;
                        }

                        if (!File.Exists(configPath))
                        {
                            result.Status = "app.config not found";
                            results.Add(result);
                            continue;
                        }

                        Configuration config = OpenConfig(configPath);

                        if (config.AppSettings == null ||
                            config.AppSettings.Settings == null ||
                            config.AppSettings.Settings.Count == 0)
                        {
                            result.Status = "IsSPOOnline/AppSettings not found";
                            results.Add(result);
                            continue;
                        }

                        string isSPOOnlineValue = GetAppSettingValue(config, "IsSPOOnline");

                        if (isSPOOnlineValue == null)
                        {
                            result.Status = "IsSPOOnline/AppSettings not found";
                            results.Add(result);
                            continue;
                        }

                        result.IsSPOOnline = isSPOOnlineValue;

                        if (string.Equals(isSPOOnlineValue.Trim(), "true", StringComparison.OrdinalIgnoreCase))
                        {
                            string appIdValue = GetAppSettingValue(config, "AppID");

                            if (appIdValue != null)
                            {
                                result.AppID = appIdValue;
                                result.Status = "Success";
                            }
                            else
                            {
                                result.Status = "AppID key not found";
                            }
                        }
                        else
                        {
                            result.Status = "Success";
                        }
                    }
                    catch (Exception ex)
                    {
                        result.Status = "Error";
                        result.ErrorMessage = ex.Message;
                    }

                    results.Add(result);
                }
            }

            WriteResultsToCsv(results, outputCsvPath);

            Console.WriteLine("Scan completed successfully.");
            Console.WriteLine("CSV file created at:");
            Console.WriteLine(outputCsvPath);
            Console.ReadLine();
        }

        private static Configuration OpenConfig(string configPath)
        {
            ExeConfigurationFileMap fileMap = new ExeConfigurationFileMap
            {
                ExeConfigFilename = configPath
            };

            return ConfigurationManager.OpenMappedExeConfiguration(fileMap, ConfigurationUserLevel.None);
        }

        private static string GetAppSettingValue(Configuration config, string keyName)
        {
            foreach (KeyValueConfigurationElement setting in config.AppSettings.Settings)
            {
                if (string.Equals(setting.Key, keyName, StringComparison.OrdinalIgnoreCase))
                {
                    return setting.Value;
                }
            }

            return null;
        }

        private static void WriteResultsToCsv(List<ConfigResult> results, string filePath)
        {
            StringBuilder sb = new StringBuilder();

            sb.AppendLine("RootPath,Mnemonic,ConfigPath,IsSPOOnline,AppID,Status,ErrorMessage");

            foreach (ConfigResult item in results)
            {
                sb.AppendLine(string.Join(",",
                    EscapeCsv(item.RootPath),
                    EscapeCsv(item.Mnemonic),
                    EscapeCsv(item.ConfigPath),
                    EscapeCsv(item.IsSPOOnline),
                    EscapeCsv(item.AppID),
                    EscapeCsv(item.Status),
                    EscapeCsv(item.ErrorMessage)
                ));
            }

            File.WriteAllText(filePath, sb.ToString(), Encoding.UTF8);
        }

        private static string EscapeCsv(string value)
        {
            if (value == null)
                return "\"\"";

            value = value.Replace("\"", "\"\"");
            return $"\"{value}\"";
        }
    }

    internal class ConfigResult
    {
        public string RootPath { get; set; }
        public string Mnemonic { get; set; }
        public string ConfigPath { get; set; }
        public string IsSPOOnline { get; set; }
        public string AppID { get; set; }
        public string Status { get; set; }
        public string ErrorMessage { get; set; }
    }
}
