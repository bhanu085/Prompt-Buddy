##############################################################
# FUNCTION: Load CSV in batch (FAST)
##############################################################

function Load-CsvInBatches {
    param(
        [string] $FilePath,
        [string] $Server,
        [string] $Database,
        [string] $Table,
        [int]    $BatchSize = 50000
    )

    # Build SQL connection
    $connString = "Server=$Server;Database=$Database;Integrated Security=SSPI"
    $connection = New-Object System.Data.SqlClient.SqlConnection($connString)
    $connection.Open()

    # Load SQL columns using SqlCommand
    $cmd = $connection.CreateCommand()
    $cmd.CommandText = @"
SELECT COLUMN_NAME
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = 'AuditLog' AND TABLE_SCHEMA = 'dbo'
"@

    $reader = $cmd.ExecuteReader()
    $sqlColumns = @()

    while ($reader.Read()) {
        $sqlColumns += $reader["COLUMN_NAME"]
    }
    $reader.Close()

    # Configure SqlBulkCopy
    $bulkCopy = New-Object System.Data.SqlClient.SqlBulkCopy($connection)
    $bulkCopy.DestinationTableName = $Table
    $bulkCopy.BatchSize = $BatchSize
    $bulkCopy.BulkCopyTimeout = 0

    # CSV parser
    $parser = New-Object Microsoft.VisualBasic.FileIO.TextFieldParser($FilePath)
    $parser.TextFieldType = "Delimited"
    $parser.SetDelimiters(",")

    # CSV header
    $headers = $parser.ReadFields()

    # Create DataTable only for matching columns
    $dt = New-Object System.Data.DataTable

    foreach ($h in $headers) {
        $clean = $h.Trim()
        if ($sqlColumns -contains $clean) {
            $null = $dt.Columns.Add($clean)
            $null = $bulkCopy.ColumnMappings.Add($clean, $clean)
        }
    }

    $rowCount = 0

    # Read CSV rows
    while (-not $parser.EndOfData) {
        $fields = $parser.ReadFields()

        $row = $dt.NewRow()

        # Fill ONLY matched columns
        foreach ($col in $dt.Columns) {
            $row[$col.ColumnName] = $fields[$headers.IndexOf($col.ColumnName)]
        }

        $dt.Rows.Add($row)
        $rowCount++

        if ($dt.Rows.Count -ge $BatchSize) {
            $bulkCopy.WriteToServer($dt)
            $dt.Clear()
        }
    }

    # Final push
    if ($dt.Rows.Count -gt 0) {
        $bulkCopy.WriteToServer($dt)
    }

    $parser.Close()
    $connection.Close()

    return $rowCount
}



##############################################################
# MAIN LOGIC: Get CSVs from OneDrive → Import → Move
##############################################################

# Fetch files from OneDrive folder
$files = Get-PnPFolderItem -FolderSiteRelativeUrl $CSVFolder -ItemType File |
         Where-Object { $_.Name -like "*.csv" }

"[$(Get-Date)] Found $($files.Count) CSV files." | Out-File $LogFile -Append

foreach ($file in $files) {

    $LocalFile = Join-Path $TempPath $file.Name
    $ProcessedLocalFile = Join-Path $ProcessedPath $file.Name

    "[$(Get-Date)] Downloading $($file.Name)..." | Out-File $LogFile -Append

    # Download from OneDrive
    Get-PnPFile -ServerRelativeUrl $file.ServerRelativeUrl `
                -Path $TempPath `
                -FileName $file.Name `
                -AsFile -Force

    "[$(Get-Date)] Importing $($file.Name) into SQL..." | Out-File $LogFile -Append

    try {
        $rows = Load-CsvInBatches -FilePath $LocalFile `
                                  -Server $SqlServer `
                                  -Database $SqlDatabase `
                                  -Table $SqlTable `
                                  -BatchSize $BatchSize

        "[$(Get-Date)] Completed: $($file.Name) - Rows Imported: $rows" | Out-File $LogFile -Append

        # Move processed file in OneDrive
        $destUrl = ($file.ServerRelativeUrl -replace $CSVFolder, "$CSVFolder/Processed")
        $destUrl

        Try {
            Move-PnPFile -ServerRelativeUrl $file.ServerRelativeUrl `
                         -TargetUrl $destUrl -Force
        }
        catch {
            "[$(Get-Date)] WARNING: Could not move file in OneDrive. $_" | Out-File $LogFile -Append
        }

        # Move local copy to processed archive
        Move-Item -Path $LocalFile -Destination $ProcessedLocalFile -Force

    }
    catch {
        "[$(Get-Date)] ERROR importing $($file.Name): $($_.Exception.Message)" | Out-File $LogFile -Append
        continue
    }
}

"[$(Get-Date)] === Import Job Finished ===" | Out-File $LogFile -Append
